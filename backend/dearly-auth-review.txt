from datetime import datetime, date

from pydantic import BaseModel, ConfigDict, Field, field_validator


class FutureLetterCreate(BaseModel):
    title: str = Field(min_length=1, max_length=255)
    content: str = Field(min_length=1)
    unlock_date: date

    @field_validator("unlock_date")
    @classmethod
    def validate_unlock_date(cls, value):
        if value < date.today():
            raise ValueError("Unlock date cannot be in the past")
        return value


class FutureLetterUpdate(BaseModel):
    title: str | None = Field(default=None, min_length=1, max_length=255)
    content: str | None = Field(default=None, min_length=1)
    unlock_date: date | None = None

    @field_validator("unlock_date")
    @classmethod
    def validate_unlock_date(cls, value):
        if value is not None and value < date.today():
            raise ValueError("Unlock date cannot be in the past")
        return value


class FutureLetterResponse(BaseModel):

    unlock_date: date

from fastapi import APIRouter, Depends, HTTPException, status
from sqlalchemy.orm import Session

from backend.database.connection import get_db
from backend.models import Letter
from backend.schemas.letter import (
    LetterCreate,
    LetterUpdate,
    LetterResponse
)

router = APIRouter(prefix="/letters", tags=["Letters"])


@router.post(
    "/",
    response_model=LetterResponse,
    status_code=status.HTTP_201_CREATED
)
def create_letter(
    letter_data: LetterCreate,
    db: Session = Depends(get_db)
):
    new_letter = Letter(
        title=letter_data.title,
        content=letter_data.content
    )

    db.add(new_letter)
    db.commit()
    db.refresh(new_letter)

    return new_letter


@router.get("/", response_model=list[LetterResponse])
def get_letters(db: Session = Depends(get_db)):
    return (
        db.query(Letter)
        .order_by(Letter.created_at.desc())
        .all()
    )


@router.get("/{letter_id}", response_model=LetterResponse)
def get_letter(
    letter_id: int,
    db: Session = Depends(get_db)
):
    letter = db.query(Letter).filter(
        Letter.id == letter_id
    ).first()

    if letter is None:
        raise HTTPException(status_code=404, detail="Letter not found")

    return letter


@router.patch("/{letter_id}", response_model=LetterResponse)
def update_letter(
    letter_id: int,
    letter_data: LetterUpdate,
    db: Session = Depends(get_db)
):
    letter = db.query(Letter).filter(
        Letter.id == letter_id
    ).first()

    if letter is None:
        raise HTTPException(status_code=404, detail="Letter not found")

    updates = letter_data.model_dump(exclude_unset=True)

    for field, value in updates.items():
        setattr(letter, field, value)

    db.commit()
    db.refresh(letter)

    return letter


@router.delete(
    "/{letter_id}",
    status_code=status.HTTP_204_NO_CONTENT
)
def delete_letter(
    letter_id: int,
    db: Session = Depends(get_db)
):
    letter = db.query(Letter).filter(
        Letter.id == letter_id
    ).first()

    if letter is None:
        raise HTTPException(status_code=404, detail="Letter not found")

    db.delete(letter)
    db.commit()

    return None

from datetime import date

from fastapi import APIRouter, Depends, HTTPException, Query, status
from sqlalchemy.orm import Session
from sqlalchemy import or_

from backend.database.connection import get_db
from backend.models import FutureLetter
from backend.schemas.future_letter import (
    FutureLetterCreate,
    FutureLetterUpdate,
    FutureLetterResponse
)

router = APIRouter(
    prefix="/future-me",
    tags=["Future Me"]
)


def format_future_letter(letter):
    unlocked = letter.unlock_date <= date.today()

    return FutureLetterResponse(
        id=letter.id,
        title=letter.title,
        content=letter.content if unlocked else None,
        unlock_date=letter.unlock_date,
        created_at=letter.created_at,
        updated_at=letter.updated_at,
        is_unlocked=unlocked,
        days_remaining=max(
            0,
            (letter.unlock_date - date.today()).days
        )
    )


# CREATE FUTURE LETTER
@router.post(
    "/",
    response_model=FutureLetterResponse,
    status_code=status.HTTP_201_CREATED
)
def create_future_letter(
    letter_data: FutureLetterCreate,
    db: Session = Depends(get_db)
):
    new_letter = FutureLetter(
        title=letter_data.title,
        content=letter_data.content,
        unlock_date=letter_data.unlock_date
    )

    db.add(new_letter)
    db.commit()
    db.refresh(new_letter)

    return format_future_letter(new_letter)


# GET ALL FUTURE LETTERS
@router.get("/", response_model=list[FutureLetterResponse])
def get_future_letters(
    status_filter: str = Query(
        default="all",
        alias="status",
        pattern="^(all|locked|unlocked)$"
    ),
    sort: str = Query(
        default="newest",
        pattern="^(newest|oldest)$"
    ),
    search: str | None = None,
    db: Session = Depends(get_db)
):
    today = date.today()

    query = db.query(FutureLetter)

    if status_filter == "locked":
        query = query.filter(FutureLetter.unlock_date > today)

    elif status_filter == "unlocked":
        query = query.filter(FutureLetter.unlock_date <= today)

    if search:
        search_term = f"%{search}%"

        query = query.filter(
            or_(
                FutureLetter.title.ilike(search_term),
                (
                    (FutureLetter.unlock_date <= today)
                    & FutureLetter.content.ilike(search_term)
                )
            )
        )

    if sort == "newest":
        query = query.order_by(FutureLetter.created_at.desc())
    else:
        query = query.order_by(FutureLetter.created_at.asc())

    return [
        format_future_letter(letter)
        for letter in query.all()
    ]


# READ ONE UNLOCKED FUTURE LETTER
@router.get("/{letter_id}", response_model=FutureLetterResponse)
def get_future_letter(
    letter_id: int,
    db: Session = Depends(get_db)
):
    letter = db.query(FutureLetter).filter(
        FutureLetter.id == letter_id
    ).first()

    if letter is None:
        raise HTTPException(
            status_code=404,
            detail="Future letter not found"
        )

    if letter.unlock_date > date.today():
        raise HTTPException(
            status_code=423,
            detail="This letter is still locked"
        )

    return format_future_letter(letter)


# UPDATE FUTURE LETTER
@router.patch("/{letter_id}", response_model=FutureLetterResponse)
def update_future_letter(
    letter_id: int,
    letter_data: FutureLetterUpdate,
    db: Session = Depends(get_db)
):
    letter = db.query(FutureLetter).filter(
        FutureLetter.id == letter_id
    ).first()

    if letter is None:
        raise HTTPException(
            status_code=404,
            detail="Future letter not found"
        )

    updates = letter_data.model_dump(exclude_unset=True)

    new_unlock_date = updates.get(
        "unlock_date",
        letter.unlock_date
    )

    if new_unlock_date is not None and new_unlock_date < date.today():
        raise HTTPException(
            status_code=422,
            detail="Unlock date cannot be in the past"
        )

    for field, value in updates.items():
        setattr(letter, field, value)

    db.commit()
    db.refresh(letter)

    return format_future_letter(letter)

    "image/webp": ".webp",
    "image/gif": ".gif"
}


def format_photo(photo):
    return MemoryPhotoResponse(
        id=photo.id,
        caption=photo.caption,
        original_name=photo.original_name,
        content_type=photo.content_type,
        file_size=photo.file_size,
        photo_url=f"/uploads/{photo.stored_name}",
        created_at=photo.created_at
    )


# UPLOAD A PHOTO
@router.post(
    "/photos",
    response_model=MemoryPhotoResponse,
    status_code=status.HTTP_201_CREATED
)
async def upload_photo(
    caption: str | None = Form(default=None),
    file: UploadFile = File(...),
    db: Session = Depends(get_db)
):
    if file.content_type not in ALLOWED_TYPES:
        raise HTTPException(
            status_code=400,
            detail="Only JPEG, PNG, WEBP and GIF images are allowed"
        )

    extension = ALLOWED_TYPES[file.content_type]
    stored_name = f"{uuid4().hex}{extension}"
    file_path = UPLOAD_DIR / stored_name

    file_size = 0

    try:
        with file_path.open("wb") as destination:
            while chunk := await file.read(1024 * 1024):
                file_size += len(chunk)

                if file_size > MAX_FILE_SIZE:
                    raise HTTPException(
                        status_code=413,
                        detail="Image size cannot exceed 10 MB"
                    )

                destination.write(chunk)

    except Exception:
        file_path.unlink(missing_ok=True)
        raise

    finally:
        await file.close()

    photo = MemoryPhoto(
        caption=caption,
        original_name=(file.filename or "photo")[:255],
        stored_name=stored_name,
        content_type=file.content_type,
        file_size=file_size
    )

    try:
        db.add(photo)
        db.commit()
        db.refresh(photo)
    except Exception:
        db.rollback()
        file_path.unlink(missing_ok=True)
        raise

    return format_photo(photo)
