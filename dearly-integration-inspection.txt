import { useEffect, useState } from "react";
import Desktop from "./Desktop.jsx";
import Auth from "./features/auth/Auth.jsx";
import {
  getAccessToken,
  clearAccessToken,
  getCurrentUser,
} from "./services/api";

function App() {
  const [user, setUser] = useState(null);
  const [checkingSession, setCheckingSession] = useState(true);

  useEffect(() => {
    let active = true;
    const token = getAccessToken();

    if (!token) {
      setCheckingSession(false);
      return () => {
        active = false;
      };
    }

    getCurrentUser(token)
      .then((currentUser) => {
        if (active) setUser(currentUser);
      })
      .catch(() => {
        clearAccessToken();
        if (active) setUser(null);
      })
      .finally(() => {
        if (active) setCheckingSession(false);
      });

    return () => {
      active = false;
    };
  }, []);

  function handleLogout() {
    clearAccessToken();
    setUser(null);
  }

  if (checkingSession) {
    return (
      <main className="auth-loading">
        <div>Opening your little world... ♡</div>
      </main>
    );
  }

  if (!user) {
    return <Auth onAuthenticated={setUser} />;
  }

  return <Desktop user={user} onLogout={handleLogout} />;
}

export default App;

const API_BASE_URL = (
  import.meta.env.VITE_API_URL || "http://127.0.0.1:8000"
).replace(/\/+$/, "");

const TOKEN_KEY = "dearly_access_token";

export function getAccessToken() {
  return sessionStorage.getItem(TOKEN_KEY);
}

export function saveAccessToken(token) {
  sessionStorage.setItem(TOKEN_KEY, token);
}

export function clearAccessToken() {
  sessionStorage.removeItem(TOKEN_KEY);
}

export async function apiRequest(path, options = {}) {
  const headers = new Headers(options.headers || {});

  if (options.body !== undefined) {
    headers.set("Content-Type", "application/json");
  }

  const token = Object.prototype.hasOwnProperty.call(options, "token")
    ? options.token
    : getAccessToken();

  if (token) {
    headers.set("Authorization", `Bearer ${token}`);
  }

  let response;

  try {
    response = await fetch(`${API_BASE_URL}${path}`, {
      ...options,
      headers,
    });
  } catch {
    throw new Error(
      "Unable to connect to Dearly's server. Please check that the backend is running."
    );
  }

  const data = await response.json().catch(() => null);

  if (!response.ok) {
    let message = data?.detail || "Something went wrong.";

    if (Array.isArray(message)) {
      message = message
        .map((item) => item.msg || "Invalid input")
        .join(", ");
    }

    throw new Error(message);
  }

  return data;
}

export function registerUser(userData) {
  return apiRequest("/auth/register", {
    method: "POST",
    body: JSON.stringify(userData),
    token: null,
  });
}

export function loginUser(credentials) {
  return apiRequest("/auth/login", {
    method: "POST",
    body: JSON.stringify(credentials),
    token: null,
  });
}

export function getCurrentUser(token) {
  return apiRequest("/auth/me", {
    method: "GET",
    ...(token ? { token } : {}),
  });
}

function normalizeLetter(letter) {
    return {
        id: letter.id,
        title: letter.title,
        content: letter.content,
        status: letter.status || "completed",
        createdAt: letter.created_at,
        updatedAt: letter.updated_at,
    };
}

function formatDate(date) {
    if (!date) return "";

    return new Date(date).toLocaleDateString("en-IN", {
        day: "numeric",
        month: "short",
        year: "numeric",
    });
}

const DB_NAME = "DearlyMemoriesDB";
const STORE_NAME = "memories";

function openDatabase() {
    return new Promise((resolve, reject) => {
        const request = indexedDB.open(DB_NAME, 1);

        request.onupgradeneeded = () => {
            const db = request.result;

            if (!db.objectStoreNames.contains(STORE_NAME)) {
                db.createObjectStore(STORE_NAME, { keyPath: "id" });
            }
        };

        request.onsuccess = () => resolve(request.result);
        request.onerror = () => reject(request.error);
    });
}

async function saveMemory(memory) {
    const db = await openDatabase();

    return new Promise((resolve, reject) => {
        const transaction = db.transaction(STORE_NAME, "readwrite");
        transaction.objectStore(STORE_NAME).put(memory);

        transaction.oncomplete = () => {
            db.close();
            resolve();
        };

        transaction.onerror = () => {
            db.close();
            reject(transaction.error);
        };
    });
}

async function getMemories() {
    const db = await openDatabase();

    return new Promise((resolve, reject) => {
        const request = db
            .transaction(STORE_NAME, "readonly")
            .objectStore(STORE_NAME)
            .getAll();

        request.onsuccess = () => {
            db.close();
            resolve(request.result || []);
        };

        request.onerror = () => {
            db.close();
            reject(request.error);
        };
    });
}

async function deleteMemory(id) {
    const db = await openDatabase();

    return new Promise((resolve, reject) => {
        const transaction = db.transaction(STORE_NAME, "readwrite");
        transaction.objectStore(STORE_NAME).delete(id);

        transaction.oncomplete = () => {
            db.close();
            resolve();
        };

        transaction.onerror = () => {
            db.close();
            reject(transaction.error);
        };
    });
}

from sqlalchemy import inspect, text
from sqlalchemy.engine import Engine


OWNED_TABLES = [
    "letters",
    "diary_entries",
    "future_letters",
    "memory_photos",
    "about_me",
    "user_settings",
]


def migrate_user_ownership(engine: Engine):
    """
    Add ownership columns and assign legacy records
    to the first existing account.
    """
    with engine.begin() as connection:
        for table in OWNED_TABLES:
            if not inspect(connection).has_table(table):
                continue

            columns = {
                column["name"]
                for column in inspect(connection).get_columns(table)
            }

            if "user_id" not in columns:
                connection.execute(text(
                    f"ALTER TABLE {table} "
                    "ADD COLUMN user_id INTEGER REFERENCES users(id)"
                ))

        # Add the letter status column to existing databases.
        if inspect(connection).has_table("letters"):
            letter_columns = {
                column["name"]
                for column in inspect(connection).get_columns("letters")
            }

            if "status" not in letter_columns:
                connection.execute(text(
                    "ALTER TABLE letters "
                    "ADD COLUMN status VARCHAR(20) "
                    "NOT NULL DEFAULT 'completed'"
                ))

        # One profile and settings row per user.
        if inspect(connection).has_table("about_me"):
            connection.execute(text(
                "CREATE UNIQUE INDEX IF NOT EXISTS "
                "ix_about_me_user_id ON about_me(user_id)"
            ))

        if inspect(connection).has_table("user_settings"):
            connection.execute(text(
                "CREATE UNIQUE INDEX IF NOT EXISTS "
                "ix_user_settings_user_id ON user_settings(user_id)"
            ))

        # Assign legacy records to the first existing account.
        if inspect(connection).has_table("users"):
            first_user_id = connection.execute(
                text("SELECT id FROM users ORDER BY id LIMIT 1")
            ).scalar()

            if first_user_id is not None:
                for table in OWNED_TABLES:
                    if inspect(connection).has_table(table):
                        connection.execute(
                            text(
                                f"UPDATE {table} "
                                "SET user_id = :user_id "
                                "WHERE user_id IS NULL"
                            ),
                            {"user_id": first_user_id},
                        )

@router.post("/register", response_model=UserResponse, status_code=201)
def register(
    user_data: UserRegister,
    db: Session = Depends(get_db)
):
    email = str(user_data.email).strip().lower()

    if db.query(User).filter(User.email == email).first():
        raise HTTPException(409, "An account with this email already exists")

    user = User(
        name=user_data.name.strip(),
        email=email,
        hashed_password=hash_password(user_data.password)
    )

    db.add(user)

    try:
        db.commit()
        db.refresh(user)
    except IntegrityError:
        db.rollback()
        raise HTTPException(409, "An account with this email already exists")

    return user


@router.post("/login", response_model=TokenResponse)
def login(
    login_data: UserLogin,
    db: Session = Depends(get_db)
):
    email = str(login_data.email).strip().lower()

    user = db.query(User).filter(User.email == email).first()

    if user is None or not verify_password(
        login_data.password,
        user.hashed_password
    ):
        raise HTTPException(
            status_code=401,
            detail="Incorrect email or password",
            headers={"WWW-Authenticate": "Bearer"}
        )

    if not user.is_active:
        raise HTTPException(403, "This account is inactive")

    return TokenResponse(
        access_token=create_access_token(user.id),
        expires_in=30 * 60
    )

def format_future_letter(letter):
    unlocked = letter.unlock_date <= date.today()
    return FutureLetterResponse(
        id=letter.id,
        title=letter.title,
        content=letter.content if unlocked else None,
        unlock_date=letter.unlock_date,
        created_at=letter.created_at,

@router.get("/", response_model=list[FutureLetterResponse])
def get_future_letters(
    status_filter: str = Query(
        default="all",
        alias="status",
        pattern="^(all|locked|unlocked)$"
    ),
    sort: str = Query(
        default="newest",
        pattern="^(newest|oldest)$"
    ),
    search: str | None = None,
    db: Session = Depends(get_db),
    user: User = Depends(get_current_user)
):
    today = date.today()

    query = db.query(FutureLetter).filter(
        FutureLetter.user_id == user.id
    )

    if status_filter == "locked":
        query = query.filter(FutureLetter.unlock_date > today)
    elif status_filter == "unlocked":
        query = query.filter(FutureLetter.unlock_date <= today)

    if search:
        term = f"%{search}%"
        query = query.filter(or_(
            FutureLetter.title.ilike(term),
            (
                (FutureLetter.unlock_date <= today)
                & FutureLetter.content.ilike(term)
            )
        ))

    if sort == "newest":
        query = query.order_by(FutureLetter.created_at.desc())
    else:
        query = query.order_by(FutureLetter.created_at.asc())

    return [format_future_letter(item) for item in query.all()]

UPLOAD_DIR = Path("uploads")
UPLOAD_DIR.mkdir(parents=True, exist_ok=True)

MAX_FILE_SIZE = 10 * 1024 * 1024

ALLOWED_TYPES = {
    "image/jpeg": ".jpg",
    "image/png": ".png",
    "image/webp": ".webp",
    "image/gif": ".gif"
}


def format_photo(photo):
    return MemoryPhotoResponse(
        id=photo.id,
        caption=photo.caption,
        original_name=photo.original_name,
        content_type=photo.content_type,
        file_size=photo.file_size,
        photo_url=f"/memories/photos/{photo.id}/file",
        created_at=photo.created_at
    )


@router.post(
    "/photos",
    response_model=MemoryPhotoResponse,
    status_code=status.HTTP_201_CREATED
)
async def upload_photo(
    caption: str | None = Form(default=None),
    file: UploadFile = File(...),
    db: Session = Depends(get_db),
    user: User = Depends(get_current_user)
):
    if file.content_type not in ALLOWED_TYPES:
        raise HTTPException(400, "Unsupported image type")

    stored_name = f"{uuid4().hex}{ALLOWED_TYPES[file.content_type]}"
    file_path = UPLOAD_DIR / stored_name
    file_size = 0

    try:
        with file_path.open("wb") as destination:
            while chunk := await file.read(1024 * 1024):
                file_size += len(chunk)
                if file_size > MAX_FILE_SIZE:
                    raise HTTPException(413, "Image size cannot exceed 10 MB")
                destination.write(chunk)
    except Exception:
        file_path.unlink(missing_ok=True)
        raise
    finally:
        await file.close()

    photo = MemoryPhoto(
        user_id=user.id,
        caption=caption,
        original_name=(file.filename or "photo")[:255],
        stored_name=stored_name,
        content_type=file.content_type,
        file_size=file_size
    )

    try:
        db.add(photo)
        db.commit()
        db.refresh(photo)
    except Exception:
        db.rollback()
        file_path.unlink(missing_ok=True)
        raise

    return format_photo(photo)
